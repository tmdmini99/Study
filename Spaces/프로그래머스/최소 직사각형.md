

큰 사이즈 작은 사이즈 변경 후 비교
```java
import java.util.*;
class Solution {
    public int solution(int[][] sizes) {
        int xSize = 0;
        int ySize = 0;
        
        for(int [] size : sizes){
            for(int i=0; i<size.length; i++){
                int temp = 0;
                if(size[0] < size[1]){
                    temp = size[0];
                    size[0] = size[1];
                    size[1] = temp;
                }
                xSize = Math.max(xSize,size[0]);
                ySize = Math.max(ySize,size[1]);
            }
        }
        return xSize*ySize;
    }
}
```
